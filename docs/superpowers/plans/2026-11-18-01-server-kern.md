# Server-Kern Implementation Plan (Etappe 1 von 5)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ein lauffähiger, getesteter Server mit Datenbank, Level-Logik, Login, validierendem Import und Übersicht-API, im öffentlichen Repo `lernsystem-s3`.

**Architecture:** Eine FastAPI-App (`server/app`) mit SQLite über das stdlib-Modul `sqlite3` (kein ORM). Reine Logik (Level, Import, Übersicht) liegt in eigenen Modulen ohne HTTP-Bezug; die Routen sind dünn. Die App wird über eine Factory `create_app(settings)` gebaut, damit Tests eine eigene DB-Datei bekommen.

**Tech Stack:** Python 3.13+ (lokal 3.14), FastAPI, uvicorn, pydantic v2, sqlite3, pytest, httpx (für `TestClient`), Docker Compose.

Spec: `docs/superpowers/specs/2026-11-18-lernsystem-s3-design.md`

Folge-Etappen (eigene Pläne): 2 Erst-Extraktion · 3 Frontend-Übersicht · 4 Blatt-Ablauf · 5 Deployment + Update-Automatik.

## Global Constraints

- Das Quell-Repo `dski-sources-3/` wird nie verändert und nie committet (steht in `.gitignore`).
- Im öffentlichen Repo stehen keine Geheimnisse, keine Datenbank, keine Server-IP. Geheimnisse kommen nur aus Umgebungsvariablen: `LERN_API_TOKEN`, `LERN_PASSWORT`, `LERN_SECRET`.
- Level-Schwellen exakt: 1 = 50 %, 2 = 65 %, 3 = 80 %, 4 = 95 %, 5 = 100 %. Neue Einträge starten auf Level 0. „Beherrscht" = Level ≥ 4.
- Aufstieg höchstens eine Stufe je Bewertung, nur wenn Claude-Wert UND User-Wert ≥ Schwelle des nächsten Levels.
- Bezeichner, Tabellen und API-Felder sind deutsch (wie in der Spec), ohne Umlaute.
- Jeder Endpunkt außer `/api/login` und `/api/gesund` verlangt Auth (Bearer-Token oder Session-Cookie).
- Abweichung von der Spec, bewusst: Der Lernstand (`level`, `claude_wert`, `user_wert`, `bewertet_am`) steht als Spalten in `eintraege` statt in einer eigenen Tabelle `stand` (1:1-Beziehung, eine Tabelle weniger).
- Tests laufen aus `server/` mit `.venv/bin/pytest`.

import os
from dataclasses import dataclass


@dataclass(frozen=True)
class Settings:
    db_pfad: str
    api_token: str
    passwort: str
    secret: str
    cookie_secure: bool = True

    @classmethod
    def aus_env(cls) -> "Settings":
        fehlend = [n for n in ("LERN_API_TOKEN", "LERN_PASSWORT", "LERN_SECRET") if not os.environ.get(n)]
        if fehlend:
            raise RuntimeError("Umgebungsvariablen fehlen: " + ", ".join(fehlend))
        return cls(
            db_pfad=os.environ.get("LERN_DB", "/data/lernsystem.db"),
            api_token=os.environ["LERN_API_TOKEN"],
            passwort=os.environ["LERN_PASSWORT"],
            secret=os.environ["LERN_SECRET"],
            cookie_secure=os.environ.get("LERN_COOKIE_SECURE", "1") != "0",
        )

import sqlite3
from pathlib import Path

_SCHEMA = Path(__file__).with_name("schema.sql")


def verbinden(pfad: str) -> sqlite3.Connection:
    """Öffnet die DB. isolation_level=None: Transaktionen werden explizit mit BEGIN geführt."""
    conn = sqlite3.connect(pfad, isolation_level=None, check_same_thread=False)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")
    conn.execute("PRAGMA journal_mode = WAL")
    return conn


def initialisieren(conn: sqlite3.Connection) -> None:
    conn.executescript(_SCHEMA.read_text(encoding="utf-8"))

`server/app/level.py`:

```python
"""Level-Logik. Einzige Stelle, an der Schwellen und Aufstiegsregeln stehen."""

SCHWELLEN = {1: 50, 2: 65, 3: 80, 4: 95, 5: 100}
MAX_LEVEL = 5
BEHERRSCHT_AB = 4


def neues_level(level: int, claude_wert: int, user_wert: int) -> int:
    """Eine Stufe hoch, wenn beide Werte die Schwelle des nächsten Levels erreichen."""
    for wert in (claude_wert, user_wert):
        if not 0 <= wert <= 100:
            raise ValueError(f"Wert außerhalb 0–100: {wert}")
    if level >= MAX_LEVEL:
        return level
    schwelle = SCHWELLEN[level + 1]
    if claude_wert >= schwelle and user_wert >= schwelle:
        return level + 1
    return level


def nach_starker_aenderung(level: int) -> int:
    """Die Quelle hat sich inhaltlich stark geändert: eine Stufe zurück."""
    return max(0, level - 1)


def ist_beherrscht(level: int) -> bool:
    return level >= BEHERRSCHT_AB
```

`server/app/auth.py`:

```python
"""Auth: Bearer-Token für die CLI, signiertes Session-Cookie für den Browser."""

import hashlib
import hmac
import time

from fastapi import HTTPException, Request

COOKIE = "lern_sitzung"
SITZUNG_SEKUNDEN = 30 * 24 * 3600
MAX_FEHLVERSUCHE = 5
FENSTER_SEKUNDEN = 300


def _signatur(secret: str, ablauf: str) -> str:
    return hmac.new(secret.encode(), ablauf.encode(), hashlib.sha256).hexdigest()


def cookie_erzeugen(secret: str, jetzt: float | None = None) -> str:
    ablauf = str(int((jetzt or time.time()) + SITZUNG_SEKUNDEN))
    return f"{ablauf}.{_signatur(secret, ablauf)}"


def cookie_gueltig(secret: str, wert: str | None, jetzt: float | None = None) -> bool:
    if not wert or "." not in wert:
        return False
    ablauf, signatur = wert.split(".", 1)
    if not hmac.compare_digest(signatur, _signatur(secret, ablauf)):
        return False
    return ablauf.isdigit() and int(ablauf) > (jetzt or time.time())


def gleich(a: str, b: str) -> bool:
    return hmac.compare_digest(a.encode(), b.encode())


def angemeldet(request: Request) -> None:
    settings = request.app.state.settings
    kopf = request.headers.get("authorization", "")
    if kopf.startswith("Bearer ") and gleich(kopf[7:], settings.api_token):
        return
    if cookie_gueltig(settings.secret, request.cookies.get(COOKIE)):
        return
    raise HTTPException(status_code=401, detail="Nicht angemeldet")


class LoginBremse:
    """Begrenzt Fehlversuche je Adresse (im Speicher, reicht für einen Benutzer)."""

    def __init__(self) -> None:
        self._fehler: dict[str, list[float]] = {}

    def gesperrt(self, adresse: str) -> bool:
        grenze = time.time() - FENSTER_SEKUNDEN
        versuche = [t for t in self._fehler.get(adresse, []) if t > grenze]
        self._fehler[adresse] = versuche
        return len(versuche) >= MAX_FEHLVERSUCHE

    def fehlversuch(self, adresse: str) -> None:
        self._fehler.setdefault(adresse, []).append(time.time())
```

`server/app/main.py`:

```python
import sqlite3
from collections.abc import Iterator
from contextlib import asynccontextmanager

from fastapi import Depends, FastAPI, HTTPException, Request, Response
from pydantic import BaseModel

from . import auth, db
from .config import Settings


def get_db(request: Request) -> Iterator[sqlite3.Connection]:
    conn = db.verbinden(request.app.state.settings.db_pfad)
    try:
        yield conn
    finally:
        conn.close()


class LoginIn(BaseModel):
    passwort: str


def create_app(settings: Settings | None = None) -> FastAPI:
    settings = settings or Settings.aus_env()

    @asynccontextmanager
    async def lifespan(app: FastAPI):
        conn = db.verbinden(settings.db_pfad)
        db.initialisieren(conn)
        conn.close()
        yield

    app = FastAPI(title="Lernsystem S3", lifespan=lifespan)
    app.state.settings = settings
    app.state.bremse = auth.LoginBremse()

    @app.get("/api/gesund")
    def gesund():
        return {"ok": True}

    @app.post("/api/login")
    def login(daten: LoginIn, request: Request, response: Response):
        adresse = request.headers.get("x-real-ip") or (request.client.host if request.client else "?")
        bremse: auth.LoginBremse = request.app.state.bremse
        if bremse.gesperrt(adresse):
            raise HTTPException(status_code=429, detail="Zu viele Fehlversuche, später erneut versuchen")
        if not auth.gleich(daten.passwort, settings.passwort):
            bremse.fehlversuch(adresse)
            raise HTTPException(status_code=401, detail="Falsches Passwort")
        response.set_cookie(
            auth.COOKIE,
            auth.cookie_erzeugen(settings.secret),
            max_age=auth.SITZUNG_SEKUNDEN,
            httponly=True,
            secure=settings.cookie_secure,
            samesite="lax",
        )
        return {"angemeldet": True}

    @app.post("/api/logout")
    def logout(response: Response):
        response.delete_cookie(auth.COOKIE)
        return {"angemeldet": False}

    @app.get("/api/ich", dependencies=[Depends(auth.angemeldet)])
    def ich():
        return {"angemeldet": True}

    return app
```

```python
"""Import aus der Extraktion: Upsert von Fächern, Themen und Einträgen, alles oder nichts."""

import sqlite3
from typing import Literal

from pydantic import BaseModel


class FachIn(BaseModel):
    slug: str
    name: str
    about_pfad: str | None = None
    about_hash: str | None = None


class ThemaIn(BaseModel):
    schluessel: str
    fach: str
    parent: str | None = None
    titel: str
    reihenfolge: int = 0


class EintragIn(BaseModel):
    schluessel: str
    thema: str
    art: Literal["aufgabe", "klausurfrage", "lernpunkt"]
    titel: str
    kurztext: str = ""
    quellpfad: str
    anker: str = ""
    quell_hash: str
    punkte: float | None = None


class ImportIn(BaseModel):
    quell_commit: str | None = None
    faecher: list[FachIn] = []
    themen: list[ThemaIn] = []
    eintraege: list[EintragIn] = []


class ImportFehler(Exception):
    pass


def _id(conn: sqlite3.Connection, tabelle: str, spalte: str, wert: str, wofuer: str) -> int:
    zeile = conn.execute(f"SELECT id FROM {tabelle} WHERE {spalte} = ?", (wert,)).fetchone()
    if zeile is None:
        raise ImportFehler(f"{wofuer}: unbekannter Verweis '{wert}'")
    return zeile["id"]


def importieren(conn: sqlite3.Connection, daten: ImportIn) -> dict:
    conn.execute("BEGIN")
    try:
        for f in daten.faecher:
            conn.execute(
                "INSERT INTO faecher(slug, name, about_pfad, about_hash) VALUES (?, ?, ?, ?)"
                " ON CONFLICT(slug) DO UPDATE SET name=excluded.name,"
                " about_pfad=excluded.about_pfad, about_hash=excluded.about_hash",
                (f.slug, f.name, f.about_pfad, f.about_hash),
            )
        for t in daten.themen:
            fach_id = _id(conn, "faecher", "slug", t.fach, f"Thema {t.schluessel}")
            parent_id = None
            if t.parent is not None:
                parent_id = _id(conn, "themen", "schluessel", t.parent, f"Thema {t.schluessel}")
            conn.execute(
                "INSERT INTO themen(fach_id, parent_id, schluessel, titel, reihenfolge) VALUES (?, ?, ?, ?, ?)"
                " ON CONFLICT(schluessel) DO UPDATE SET fach_id=excluded.fach_id,"
                " parent_id=excluded.parent_id, titel=excluded.titel, reihenfolge=excluded.reihenfolge",
                (fach_id, parent_id, t.schluessel, t.titel, t.reihenfolge),
            )
        for e in daten.eintraege:
            thema_id = _id(conn, "themen", "schluessel", e.thema, f"Eintrag {e.schluessel}")
            conn.execute(
                "INSERT INTO eintraege(schluessel, thema_id, art, titel, kurztext, quellpfad, anker, quell_hash, punkte)"
                " VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)"
                " ON CONFLICT(schluessel) DO UPDATE SET thema_id=excluded.thema_id, art=excluded.art,"
                " titel=excluded.titel, kurztext=excluded.kurztext, quellpfad=excluded.quellpfad,"
                " anker=excluded.anker, quell_hash=excluded.quell_hash, punkte=excluded.punkte",
                (e.schluessel, thema_id, e.art, e.titel, e.kurztext, e.quellpfad, e.anker, e.quell_hash, e.punkte),
            )
        if daten.quell_commit:
            conn.execute(
                "INSERT INTO meta(schluessel, wert) VALUES ('quell_commit', ?)"
                " ON CONFLICT(schluessel) DO UPDATE SET wert=excluded.wert",
                (daten.quell_commit,),
            )
        conn.execute("COMMIT")
    except BaseException:
        conn.execute("ROLLBACK")
        raise
    return {"faecher": len(daten.faecher), "themen": len(daten.themen), "eintraege": len(daten.eintraege)}
```

```python
"""Baum Fach → Thema → Unterthema → Einträge mit Zählern für die Übersicht."""

import sqlite3

from . import level


def _zaehler(gesamt: int, beherrscht: int, level_summe: int) -> dict:
    fortschritt = round(level_summe / (level.MAX_LEVEL * gesamt), 2) if gesamt else 0.0
    return {"gesamt": gesamt, "beherrscht": beherrscht, "fortschritt": fortschritt}


def baum(conn: sqlite3.Connection) -> list[dict]:
    eintraege_je_thema: dict[int, list[dict]] = {}
    for z in conn.execute(
        "SELECT id, thema_id, art, titel, kurztext, quellpfad, anker, punkte, aktiv, status,"
        " level, claude_wert, user_wert FROM eintraege WHERE status != 'archiviert' ORDER BY id"
    ):
        eintrag = dict(z)
        eintrag["aktiv"] = bool(eintrag["aktiv"])
        eintraege_je_thema.setdefault(eintrag.pop("thema_id"), []).append(eintrag)

    kinder: dict[tuple[int, int | None], list[sqlite3.Row]] = {}
    for z in conn.execute("SELECT id, fach_id, parent_id, titel, aktiv FROM themen ORDER BY reihenfolge, id"):
        kinder.setdefault((z["fach_id"], z["parent_id"]), []).append(z)

    def knoten(zeile: sqlite3.Row) -> tuple[dict, int, int, int]:
        eintraege = eintraege_je_thema.get(zeile["id"], [])
        aktive = [e for e in eintraege if e["aktiv"]]
        gesamt = len(aktive)
        beherrscht = sum(1 for e in aktive if level.ist_beherrscht(e["level"]))
        summe = sum(e["level"] for e in aktive)
        unterthemen = []
        for kind in kinder.get((zeile["fach_id"], zeile["id"]), []):
            unter, g, b, s = knoten(kind)
            unterthemen.append(unter)
            gesamt, beherrscht, summe = gesamt + g, beherrscht + b, summe + s
        if not zeile["aktiv"]:
            gesamt = beherrscht = summe = 0
        thema = {
            "id": zeile["id"],
            "titel": zeile["titel"],
            "aktiv": bool(zeile["aktiv"]),
            **_zaehler(gesamt, beherrscht, summe),
            "themen": unterthemen,
            "eintraege": eintraege,
        }
        return thema, gesamt, beherrscht, summe

    ergebnis = []
    for fach in conn.execute("SELECT id, slug, name FROM faecher ORDER BY name"):
        gesamt = beherrscht = summe = 0
        themen = []
        for wurzel in kinder.get((fach["id"], None), []):
            thema, g, b, s = knoten(wurzel)
            themen.append(thema)
            gesamt, beherrscht, summe = gesamt + g, beherrscht + b, summe + s
        ergebnis.append({
            "id": fach["id"], "slug": fach["slug"], "name": fach["name"],
            **_zaehler(gesamt, beherrscht, summe), "themen": themen,
        })
    return ergebnis
```
